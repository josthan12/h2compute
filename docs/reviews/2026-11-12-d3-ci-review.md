Exact link/JSON script, passed as a PowerShell single-quoted here-string to
`python -`:

```python
import json,re,subprocess
from pathlib import Path
root=Path.cwd(); base='c7cc59ad394c61a9e822183bd2d7e34791dc869c'; candidate='70fc5e6b93214e6222464579f85793c153a654a7'
names=subprocess.check_output(['git','diff','--name-only',base,candidate],encoding='utf-8').splitlines()
links=[]; missing=[]; json_files=[]
for name in names:
    content=subprocess.check_output(['git','show',candidate+':'+name],encoding='utf-8')
    if name.endswith('.json'): json.loads(content); json_files.append(name)
    if not name.endswith('.md'): continue
    for target in re.findall(r'(?<!!)\[[^\]]+\]\(([^)]+)\)',content):
        if target.startswith(('http:','https:','#','mailto:')): continue
        target=target.split('#',1)[0]
        if not target: continue
        links.append((name,target))
        if not (root/name).parent.joinpath(target).exists(): missing.append((name,target))
print(json.dumps({'changed_paths':len(names),'relative_targets':len(links),'missing':missing,'json_files':json_files}))
raise SystemExit(bool(missing))
```
